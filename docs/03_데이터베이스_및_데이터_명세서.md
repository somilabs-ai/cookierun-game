# 🗄️ [데이터 명세서] 쿠키 데이터 구조 & Supabase 스키마

---

## 1. 쿠키 메타데이터 객체 명세 (JSON Schema)

```typescript
export interface CookieData {
  id: string;                  // 고유 ID (예: 'pure_vanilla')
  name: string;                // 쿠키 이름 (예: '퓨어바닐라 쿠키')
  rarity: 'COMMON' | 'RARE' | 'EPIC' | 'SUPER_EPIC' | 'LEGENDARY' | 'ANCIENT' | 'BEAST';
  classType: '돌격형' | '방어형' | '침투형' | '마법형' | '사격형' | '폭발형' | '지원형' | '회복형';
  position: '전방' | '중앙' | '후방';
  element: '무속성' | '빛' | '어둠' | '불' | '물' | '얼음' | '바람' | '대지' | '번개';
  releaseYear: number;         // 출시 년도 (예: 2021)
  initialConsonants: string;   // 초성 (예: 'ㅍㅇㅂㄴㄹ ㅋㅋㅋ')
  quote: string;               // 대표 대사
  skillName: string;           // 스킬 이름
  imageUrl: string;            // 이미지 일러스트 경로
  silhouetteUrl?: string;      // 실루엣 이미지 경로
}
```

## 2. Supabase 데이터베이스 테이블 명세

```sql
-- 1. 쿠키 정보 테이블
CREATE TABLE cookies (
    id VARCHAR(50) PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    rarity VARCHAR(30) NOT NULL,
    class_type VARCHAR(30) NOT NULL,
    position VARCHAR(20) NOT NULL,
    element VARCHAR(30) DEFAULT '무속성',
    release_year INT NOT NULL,
    initial_consonants VARCHAR(100) NOT NULL,
    quote TEXT,
    skill_name VARCHAR(100),
    image_url TEXT NOT NULL,
    silhouette_url TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. 게임 스코어 및 랭킹 테이블
CREATE TABLE game_leaderboard (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nickname VARCHAR(50) NOT NULL,
    game_mode VARCHAR(50) NOT NULL,
    score INT NOT NULL,
    max_streak INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. 이상형 월드컵 투표 통계 테이블
CREATE TABLE worldcup_votes (
    cookie_id VARCHAR(50) PRIMARY KEY REFERENCES cookies(id),
    win_count INT DEFAULT 0,
    match_count INT DEFAULT 0,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```
